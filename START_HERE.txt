🔧 CUSTOM SCRIPT - Full Control
--------------------------------------------------------------------------------
Create my_run.py:

from autogluon_imbalanced_model import *

# Load your data
train_data, test_data = load_and_prepare_data('data.csv', 'target')

# Train (30 min, good quality)
predictor = train_autogluon_model(
    train_data, 
    time_limit=1800, 
    preset='good_quality',
    eval_metric='average_precision'
)

# Evaluate
results = evaluate_model(predictor, test_data)
optimal_threshold = find_optimal_threshold(results, metric='f1')
results = evaluate_model(predictor, test_data, threshold=optimal_threshold)
plot_evaluation_metrics(results)

Run: python my_run.py


🎯 ADVANCED - Best Results with SMOTE
--------------------------------------------------------------------------------
Create my_advanced_run.py:

from advanced_autogluon_pipeline import ImbalancedClassificationPipeline
import pandas as pd
from sklearn.model_selection import train_test_split

df = pd.read_csv('data.csv')
train_data, test_data = train_test_split(df, test_size=0.2, stratify=df['target'])

pipeline = ImbalancedClassificationPipeline(target_column='target')
predictor = pipeline.train_model(
    train_data, 
    time_limit=1800,
    preset='good_quality',
    resampling_method='smote'  # Uses SMOTE for better minority class handling
)

results = pipeline.comprehensive_evaluation(test_data)
pipeline.plot_comprehensive_results()
pipeline.save_results()

Run: python my_advanced_run.py
